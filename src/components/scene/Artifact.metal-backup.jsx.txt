// scene/Artifact.jsx
"use client"

import React, { forwardRef, useEffect, useMemo, useRef } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import { Center, Environment, useGLTF } from "@react-three/drei";
import * as THREE from "three";
import {
    frameViewport, CORE_REFORM, CORE_SCALE,
} from "@/components/scene/servicesData";
import { RIM_VERTEX, RIM_FRAGMENT } from "@/components/scene/rift/riftShaders";

// Dark iridescent metal: near-black brushed gunmetal with a thin-film
// sheen that drifts teal -> violet as the logo tilts. It reflects its own
// copy of the nebula (NEBULA_ENV), so it picks up the scene's colours.
// Only uniform-backed properties are animated below (iridescence and
// clearcoat stay on at a fixed amount = no shader recompile mid-scroll).
const NEBULA_ENV = "/hdri/Nebula_N0.PNG";

// Metal at rest (matches the `metal` props below)
const BASE_METAL = {
    color: new THREE.Color("#2b323d"), roughness: 0.3, filmMax: 420, envIntensity: 1.6,
};

// Subtle per-service variation while the logo acts as the orbit core,
// indexed like SERVICES: ai, software, web, marketing, design.
// filmMax shifts the thin-film hue: lower = teal/green, higher = violet/magenta.
const SERVICE_METAL = [
    // AI: restless, cooler sheen
    { color: new THREE.Color("#26323c"), roughness: 0.26, filmMax: 360, envIntensity: 1.8 },
    // Software: clean, tighter highlights
    { color: new THREE.Color("#2b323d"), roughness: 0.2,  filmMax: 420, envIntensity: 1.6 },
    // Web: liquid, glossier
    { color: new THREE.Color("#233540"), roughness: 0.16, filmMax: 380, envIntensity: 1.9 },
    // Marketing: warmer violet/magenta film
    { color: new THREE.Color("#2e2c3d"), roughness: 0.26, filmMax: 560, envIntensity: 1.7 },
    // Design: more satin, heavier
    { color: new THREE.Color("#2a3140"), roughness: 0.38, filmMax: 480, envIntensity: 1.5 },
];
const tmpColor = new THREE.Color();

// Void mode (rift.dark = 1): the world goes black, so the metal catches
// light on its own: glossier, brighter reflections, a faint inner glow.
// Every value below is the material's default at dark = 0.
const VOID_EMISSIVE = new THREE.Color("#0b0f2e");   // faint inner blue-violet
const EDGE_COLOR    = new THREE.Color("#6fdcff").multiplyScalar(2.2); // HDR: catches Bloom

const Artifact = forwardRef(function Artifact(
    { mood = null, freeze = false, scrollProgress = 0, activeService = 0, rift = null },
    externalRef
) {
    const internalRef = useRef();
    const group = externalRef ?? internalRef;

    // One ref slot per mesh — used to animate the metal in useFrame
    const materialRefs = useRef([]);
    const spinRef = useRef(null); // inner group: slow spin while acting as core
    const rimRefs    = useRef([]);
    const edgeRefs   = useRef([]);
    const ambientRef = useRef();
    const spotRef    = useRef();

    // frameViewport: identical to `viewport` at the landing camera, and still
    // correct while the camera chases the logo during the rift
    const size = useThree((s) => s.size);
    const responsiveScale = Math.min(1.5, Math.max(0.65, frameViewport(size).width * 0.22));

    const { scene } = useGLTF("/models/aura-logo.glb");

    // The logo's own environment map: the nebula itself, so reflections
    // match the backdrop. Small PNG (not the 69 MB HDR); three converts it
    // for rough reflections once it has loaded. If it is missing, the metal
    // falls back to the scene environment instead of failing.
    const envMap = useMemo(() => {
        const t = new THREE.TextureLoader().load(
            NEBULA_ENV,
            undefined,
            undefined,
            () => console.warn(`${NEBULA_ENV} not found; logo uses the scene environment.`)
        );
        t.mapping = THREE.EquirectangularReflectionMapping;
        t.colorSpace = THREE.SRGBColorSpace;
        return t;
    }, []);
    useEffect(() => () => envMap.dispose(), [envMap]);

    const metal = useMemo(
        () => ({
            color: BASE_METAL.color,
            metalness: 1,
            roughness: BASE_METAL.roughness,
            envMap,
            envMapIntensity: BASE_METAL.envIntensity,
            // No anisotropy: the logo's UVs are degenerate on the extruded sides, so
            // three's anisotropic shading divides by zero there -> NaN pixels, and
            // Bloom smears those NaNs across the whole frame (black screen).
            iridescence: 1,
            iridescenceIOR: 1.35,
            iridescenceThicknessRange: [120, BASE_METAL.filmMax],
            clearcoat: 0.35,                  // thin lacquer: a crisp top highlight
            clearcoatRoughness: 0.12,
            emissive: "#000000",
        }),
        [envMap]
    );

    const meshes = useMemo(() => {
        const arr = [];
        scene.updateMatrixWorld(true);
        scene.traverse((obj) => {
            if (!obj.isMesh) return;
            arr.push({ geometry: obj.geometry, matrix: obj.matrixWorld.clone() });
        });
        return arr;
    }, [scene]);

    // Hard edges of the logo for the void-mode glints (bevels, not every
    // triangle seam)
    const edges = useMemo(
        () => meshes.map((m) => new THREE.EdgesGeometry(m.geometry, 28)),
        [meshes]
    );

    const rimUniforms = useMemo(() => ({
        uDark: { value: 0 },
        uTime: { value: 0 },
    }), []);

    useFrame((state, delta) => {
        if (!group.current) return;

        const sp = scrollProgress;

        // SCROLL PHASE 4 (0.55-0.70): logo reforms as the orbit core
        const coreT = THREE.MathUtils.smoothstep(sp, CORE_REFORM.start, CORE_REFORM.end);
        // Spin lives on the inner group. Before the core phase it eases back to
        // the nearest full turn (the logo stays visible through the rift now),
        // so the landing always shows it face-on.
        if (spinRef.current) {
            const spin = spinRef.current;
            if (sp < CORE_REFORM.start) {
                const rest = Math.round(spin.rotation.y / (Math.PI * 2)) * Math.PI * 2;
                spin.rotation.y = THREE.MathUtils.damp(spin.rotation.y, rest, 3, delta);
            } else if (!freeze) {
                spin.rotation.y += delta * 0.35 * coreT;
            }
        }

        // QUANTUM RIFT pull: the rift grips the logo (smooth tremble, two
        // detuned sines, not random jitter) and tips it back into the void.
        // Both are 0 on the landing.
        if (spinRef.current && rift) {
            const t = state.clock.elapsedTime;
            const g = rift.pull * 0.022;
            spinRef.current.position.set(
                (Math.sin(t * 23.1) * 0.6 + Math.sin(t * 37.7 + 1.3) * 0.4) * g,
                (Math.sin(t * 29.3 + 0.7) * 0.6 + Math.sin(t * 41.9 + 2.1) * 0.4) * g,
                0
            );
            spinRef.current.rotation.x = -0.45 * rift.tilt;
            spinRef.current.rotation.z = 0.12 * rift.tilt;
        }

        // SCROLL PHASE 1 (0.10-0.25): pointer sensitivity ramps up 0.5 → 0.8
        // SCROLL PHASE 2 (0.25-0.45): sensitivity returns 0.8 → 0.5
        const sensitivity =
            sp < 0.10 ? 0.5
            : sp < 0.25 ? THREE.MathUtils.lerp(0.5, 0.8, (sp - 0.10) / 0.15)
            : sp < 0.45 ? THREE.MathUtils.lerp(0.8, 0.5, (sp - 0.25) / 0.20)
            : 0.5;

        // Pointer-reactive rotation
        if (state.pointer && !freeze) {
            const { x, y } = state.pointer;
            group.current.rotation.y = THREE.MathUtils.lerp(
                group.current.rotation.y,
                x * sensitivity,
                0.05
            );
            group.current.rotation.x = THREE.MathUtils.lerp(
                group.current.rotation.x,
                -y * sensitivity,
                0.05
            );
        }

        // Gentle floating animation
        if (!freeze) {
            const targetY = Math.sin(state.clock.elapsedTime * 0.5) * 0.04;
            group.current.position.y = THREE.MathUtils.lerp(
                group.current.position.y,
                targetY,
                0.02
            );
        }

        // SCROLL PHASE 4: blend toward the selected service's metal preset
        const preset = SERVICE_METAL[activeService] ?? SERVICE_METAL[0];
        const k = 1 - Math.exp(-delta * 2.5);
        const mix = (key) => BASE_METAL[key] + (preset[key] - BASE_METAL[key]) * coreT;
        tmpColor.copy(BASE_METAL.color).lerp(preset.color, coreT);

        // QUANTUM RIFT void mode: dark = 0 on the landing (all defaults),
        // 1 once the camera has crossed into the void.
        const dark = rift ? rift.dark : 0;

        materialRefs.current.forEach((mat) => {
            if (!mat) return;
            mat.color.lerp(tmpColor, k);
            mat.roughness = THREE.MathUtils.lerp(
                mat.roughness, mix("roughness") * (1 - 0.45 * dark), k   // glossier in the void
            );
            mat.iridescenceThicknessRange[1] = THREE.MathUtils.lerp(
                mat.iridescenceThicknessRange[1], mix("filmMax"), k
            );
            mat.envMapIntensity = mix("envIntensity") * (1 + 1.2 * dark); // brighter reflections
            mat.emissive.copy(VOID_EMISSIVE).multiplyScalar(dark);
        });

        // Fresnel rim + edge glints fade in with the darkness
        // (all rim meshes share one uniforms object, so set it via the first)
        const rimU = rimRefs.current[0]?.material.uniforms;
        if (rimU) {
            rimU.uDark.value = dark;
            rimU.uTime.value = state.clock.elapsedTime;
        }
        rimRefs.current.forEach((m) => { if (m) m.visible = dark > 0.001; });
        edgeRefs.current.forEach((l) => {
            if (!l) return;
            l.visible = dark > 0.001;
            l.material.opacity = 0.55 * dark;
        });

        // Ambient + spot dial down to empty-void lighting
        if (ambientRef.current) ambientRef.current.intensity = 0.5 * (1 - dark);
        if (spotRef.current)    spotRef.current.intensity = 1 - dark;

        // QUANTUM RIFT: the logo stays where it is (no pull into the portal)
        // and vanishes in place as the plunge plays (rift.logoScale 1 -> 0).
        let targetScale;
        if (rift) {
            targetScale = rift.logoScale;
            group.current.position.z = rift.logoZ;
        } else {
            targetScale =
                sp < 0.25 ? 1.0
                : sp < 0.45 ? THREE.MathUtils.lerp(1.0, 0.0, (sp - 0.25) / 0.20)
                : CORE_SCALE * coreT;
        }

        group.current.scale.setScalar(
            THREE.MathUtils.lerp(group.current.scale.x, targetScale, 0.08)
        );
    });

    return (
        <group ref={group} position={[0, 0, 0]}>
            <Environment
                preset="night"
                background={false}
                rotation={[Math.PI / 2, 0, 0]}
                environmentIntensity={0.01}
                resolution={256}
            />

            <ambientLight ref={ambientRef} intensity={0.5} />
            <spotLight ref={spotRef} position={[10, 10, 10]} angle={0.15} penumbra={1} />

            <group ref={spinRef}>
                <Center scale={responsiveScale}>
                    {meshes.map((m, i) => (
                        <mesh
                            key={`metal-${i}`}
                            geometry={m.geometry}
                            matrix={m.matrix}
                            matrixAutoUpdate={false}
                            frustumCulled={false}
                        >
                            <meshPhysicalMaterial
                                ref={(r) => (materialRefs.current[i] = r)}
                                {...metal}
                            />
                        </mesh>
                    ))}

                    {/* Void mode only: Fresnel rim over the metal (additive) */}
                    {meshes.map((m, i) => (
                        <mesh
                            key={`rim-${i}`}
                            ref={(r) => (rimRefs.current[i] = r)}
                            geometry={m.geometry}
                            matrix={m.matrix}
                            matrixAutoUpdate={false}
                            frustumCulled={false}
                            visible={false}
                            renderOrder={1}
                        >
                            <shaderMaterial
                                vertexShader={RIM_VERTEX}
                                fragmentShader={RIM_FRAGMENT}
                                uniforms={rimUniforms}
                                transparent
                                depthWrite={false}
                                blending={THREE.AdditiveBlending}
                                toneMapped={false}
                            />
                        </mesh>
                    ))}

                    {/* Void mode only: edge glints along the logo's bevels */}
                    {edges.map((g, i) => (
                        <lineSegments
                            key={`edge-${i}`}
                            ref={(r) => (edgeRefs.current[i] = r)}
                            geometry={g}
                            matrix={meshes[i].matrix}
                            matrixAutoUpdate={false}
                            frustumCulled={false}
                            visible={false}
                            renderOrder={1}
                        >
                            <lineBasicMaterial
                                color={EDGE_COLOR}
                                transparent
                                opacity={0}
                                depthWrite={false}
                                blending={THREE.AdditiveBlending}
                                toneMapped={false}
                            />
                        </lineSegments>
                    ))}
                </Center>
            </group>
        </group>
    );
});

export default Artifact;

useGLTF.preload("/models/aura-logo.glb");
